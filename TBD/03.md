# Clase 03. Cursores

```sql
DECLARE
	CURSOR c_creditos IS
		SELECT
			nro_solic_credito,
			monto_credito,
			total_cuotas_credito
		from credito_cliente
		ORDER BY nro_solic_credito;

	v_fila c_creditos%rowtype;
BEGIN
	OPEN c_creditos;
	LOOP
		FETCH c_creditos INTO v_fila;
			EXIT WHEN c_creditos%notfound;
			dbms_output.put_line(v_fila.nro_solic_credito || ' -> ' ||v_fila.monto_credito);
	END LOOP;
	CLOSE c_creditos;
END;
```

```sql
DECLARE
	CURSOR c_creditos IS
		SELECT nro_solic_credito, monto_credito
		FROM credito_cliente
		ORDER BY nro_solic_credito;
BEGIN
	FOR v_fila IN c_creditos LOOP
		dbms_output.put_line(v_fila.nro_solic_credito||' -> '||v_fila.monto_credito);
	END LOOP;
END;
```

```sql
DECLARE
	CURSOR c_por_sucursal (p_sucursal NUMBER) IS
		SELECT nro_solic_credito, monto_credito
		FROM credito_cliente
		WHERE id_sucursal = 2011
		ORDER BY nro_solic_credito;
BEGIN
	FOR v_fila IN c_por_sucursal(2011) LOOP
		dbms_output.put_line(v_fila.nro_solic_credito);
	END LOOP;

	FOR v_fila IN c_por_sucursal(2011) LOOP
		dbms_output.put_line(v_fila.nro_solic_credito);

	END LOOP;
END;
```
