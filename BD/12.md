# Bloques anónimos

```sql
SELECT * FROM empleado;

SELECT
    initcap(nombre_emp||' '||appaterno_emp||' '||APMATERNO_EMP) AS nombre
    ,numrut_emp||'-'||DVRUT_EMP AS rut
    ,sueldo_emp AS sueldo
FROM empleado
WHERE numrut_emp = 11846972 ---18560875
;

DECLARE
	-- variables para uso de la query
	v_nombre varchar(255);
	v_rut varchar(255);
	v_sueldo empleado.sueldo_emp%TYPE;

	-- variables del negocio
	v_bonif_extra number(10);
BEGIN
	SELECT
		nombre_emp||' '||appaterno_emp||' '||APMATERNO_EMP AS nombre
		,numrut_emp||'-'||DVRUT_EMP AS rut
		,sueldo_emp AS sueldo
	INTO
	  v_nombre
	  ,v_rut
	  ,v_sueldo
	FROM empleado
	WHERE numrut_emp = 11846972 ---18560875
	;

	-- Lógica de negocio
	v_bonif_extra := v_sueldo * 0.4;

	DBMS_OUTPUT.put_line('DATOS CÁLCULO BONIFICACIÓN EXTRA DEL 40% DEL SUELDO');
	DBMS_OUTPUT.put_line('Nombre Empleado: ' || v_nombre);
	DBMS_OUTPUT.put_line('RUN: ' || v_rut);
	DBMS_OUTPUT.put_line('Sueldo: ' || v_sueldo);
	DBMS_OUTPUT.put_line('Bonificación Extra: ' || v_bonif_extra);
END
;
```
